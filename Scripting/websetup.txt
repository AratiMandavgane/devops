#!/bin/bash
echo "Installing httpd,wget,unzip packages"
sudo yum install httpd wget unzip -y > /dev/null

echo"starting and enabling the httpd service"

sudo systemctl start httpd
sudo systemctl enable httpd

echo"starting the deployment"
mkdir -p /tmp/webfiles
cd /tmp/webfiles
echo 

wget https://www.tooplate.com/zip-templates/2160_exhibit_studio.zip > /dev/null
unzip 2160_exhibit_studio.zip > /dev/null
sudo cp -r 2160_exhibit_studio/* /var/www/html

echo 
echo "restarting httpd service"

systemctl restart httpd
rm -rf /tmp/webfiles

sudo systemctl status httpd
ls /var/www/html
